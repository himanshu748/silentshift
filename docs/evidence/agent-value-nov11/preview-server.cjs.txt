const http=require('node:http'),fs=require('node:fs/promises'),path=require('node:path');
const roots={before:path.resolve('../tour/docs/site'),after:path.resolve('docs/site')};
http.createServer(async(req,res)=>{
 try{const u=new URL(req.url,'http://local'),parts=u.pathname.split('/'),which=parts[1],root=roots[which];if(!root){res.writeHead(404);res.end('Not found');return;}
 const target=path.resolve(root,'.'+decodeURIComponent('/'+parts.slice(2).join('/')));if(target!==root&&!target.startsWith(root+path.sep))throw Error('outside root');const st=await fs.stat(target),file=st.isDirectory()?path.join(target,'index.html'):target;
 const b=await fs.readFile(file);res.writeHead(200,{'Content-Type':{'.html':'text/html','.css':'text/css','.mjs':'text/javascript','.js':'text/javascript','.json':'application/json','.woff2':'font/woff2','.png':'image/png','.webp':'image/webp','.jpg':'image/jpeg','.vtt':'text/vtt'}[path.extname(file)]||'application/octet-stream','Content-Length':b.length});res.end(b);
 }catch{res.writeHead(404);res.end('Not found');}
}).listen(5202,'127.0.0.1',()=>console.log('Loopback comparison server: /before/ and /after/'));
